Version = 1.0
Pip = 1

[Field = InputFile]
short = input
type = FNM
usage = Input file to analyze for tilt angle in the filename
tooltip = 
manpage = Input file to analyze for tilt angle in the filename.  Non-option
arguments will also be used for input files, with those entries used after any
names entered with this option.

[Field = ListOfInputFiles]
short = listin
type = FN
usage = File with list of filenames to sort, instead of names on command line
tooltip = 
manpage = File with list of filenames to sort.  You can not enter some names
with this option and other names with \fB-input\fR or as command-line
arguments; names must be entered one way or the other.

[Field = TiltSeriesFile]
short = stack
type = FN
usage = Tilt series stack file with tilt angles to match in extended header
tooltip = 
manpage = Tilt series stack file with tilt angles that can be extracted from the
extended header.  Filenames matching those angles will be put out in
the same order as those angles.

[Field = TiltAngleFile]
short = angle
type = FN
usage = Text file with tilt angles to match, one per line
tooltip = 
manpage = Text file with tilt angles, one per line.  Filenames matching those
angles will be put out in the same order as those angles.

[Field = OutputFileList]
short = outlist
type = FN
usage = Output file for list of sorted filenames
tooltip = 
manpage = Output file for list of sorted or unsorted filenames.  This file can
be used with the \fB-list\fR option to Alignframes(1).

[Field = OutputTiltAngleFile]
short = tilt
type = FN
usage = Output file for tilt angles, unsorted unless -o entered also
tooltip = 
manpage = Output file for list of sorted or unsorted tilt angles extracted from
the filenames.  This file can be used with
the \fB-tilt\fR option to Alignframes(1).  This option can be used only
if there are no tilt series angles to match.

[Field = ReverseOrder]
short = reverse
type = B
usage = Reverse order from positive to negative (if no angles to match)
tooltip = 
manpage = Reverse order from positive to negative.  This option can be used only
if there are no tilt series angles to match.

[Field = UnsortedOutput]
short = unsorted
type = B
usage = Output unsorted lists (if no angles to match)
tooltip = 
manpage = Produce unsorted output lists.  This option cannot be used with
\fB-reverse\fB, \fB-series\fB, or \fB-angle\fB.

[Field = Delimiters]
short = delim
type = CH
usage = Starting and ending delimiters of tilt angle (2 characters, default "[]")'
tooltip = 
manpage = Starting and ending delimiters of tilt angle in the filenames.  Two
characters must be entered, even if they are the same.  The default is
"[]" for tilt angles enclosed in brackets.  Files from SerialEM would
require either "__" or "_."

[Field = FixedImageDose]
short = fixed
type = F
usage = Fixed dose for each image in electrons per square Angstrom
tooltip = 
manpage = Fixed dose for each image in electrons per square Angstrom.  The
\fB-dose\fR option must also be entered.

[Field = DoseOutputFile]
short = dose
type = FN
usage = Output dose information file of type 2 (accumulated dose and image
dose)
tooltip = 
manpage = Output file for dose information in the type 2 accepted by
Alignframes(1) and Mtffilter(1), namely the prior accumulated dose and image dose for
each tilt series image.  The \fB-fixed\fR option must also be entered.  These
options are appropriate if input files are in the order of acquisition and
unsorted output is being obtained.

[Field = usage]
short = help
long = usage
type = B
usage = Print help output
tooltip = 
manpage = Print help output.
