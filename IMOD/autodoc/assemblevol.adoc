Version=1.0
Pip=1

[Field = InputFile]
short = input
type = FNM
usage = Input file
tooltip =
manpage = Input file.  Files must be entered in the order of left to right in
X, then bottom to top in Y, then lower to upper in Z.  For example, if there
are files only in X and Y, enter all of the files in the first row in Y in
order by increasing X, then the files in the second row, etc.  Non-option
arguments will also be taken as input files after the ones specified by this
option.

[Field = OutputFile]
short = output
type = FN
usage = Output file for assembled volume
tooltip =
manpage = Output file for assembled volume.  If this entry is omitted, the
last non-option argument will be used for the output filename.

[Field = StartEndToExtractInX]
short = xextract
type = IPM
usage = Starting and ending coordinates to extract at one file position in X
tooltip =
manpage = Starting and ending coordinates to extract at one file position in
X, numbered from 0, or 0,0 to use all pixels in X, or -1,-1 to use just the
first pixel in X.  If there is more than one
file position in X, this option must be entered for each position in X, unless
the number of files in X is entered with the \fB-nxfiles\fR option and all
pixels are to be taken from each file. If neither this option nor
\fB-nxfiles\fR is entered, the default is to assume one position in X with all
pixels used.  Note that the use of 0,0 to specify using all pixels requires
the different special entry of -1,-1 to use just the first pixel when the size
in X is greater than 1.

[Field = StartEndToExtractInY]
short = yextract
type = IPM
usage = Starting and ending coordinates to extract at one file position in Y
tooltip =
manpage = Starting and ending coordinates to extract at one file position in
Y, numbered from 0, or 0,0 to use all pixels in Y.  The same rules apply to
this option as to \fR-xextract\fR.

[Field = StartEndToExtractInZ]
short = zextract
type = IPM

tooltip =
manpage = Starting and ending coordinates to extract at one file position in
Z, numbered from 0, or 0,0 to use all pixels in Z.  The same rules apply to
this option as to \fR-xextract\fR.

[Field = NumberOfFilesInX]

tooltip =
